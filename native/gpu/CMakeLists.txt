cmake_minimum_required(VERSION 3.25)
project(mis_cuda_service LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(puzzlescript-guided-service cuda/production-service.cu "${ENGINE_ROOT}/native/src/runtime/json.cpp" "${ENGINE_ROOT}/native/third_party/simdjson/simdjson.cpp")
target_include_directories(puzzlescript-guided-service PRIVATE "${ENGINE_ROOT}/native/src" "${ENGINE_ROOT}/native/third_party/simdjson")
target_compile_definitions(puzzlescript-guided-service PRIVATE PS_CUDA_FORCE_INLINE=1)
target_compile_options(puzzlescript-guided-service PRIVATE $<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:--ptxas-options=-v;-lineinfo> $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CXX_COMPILER_ID:MSVC>>:-Xcompiler=/O2;-Xcompiler=/bigobj;-Xcompiler=/Zc:preprocessor>)
